Private Sub Worksheet_Change(ByVal Target As Range)
    Dim lastRouteRow As Long
    Dim watchedRange As Range

    If Target.CountLarge > 500 Then Exit Sub

    lastRouteRow = Me.Cells(Me.Rows.Count, "B").End(xlUp).Row
    If lastRouteRow < 2 Then lastRouteRow = 2

    Set watchedRange = Me.Range("A2:G" & lastRouteRow)
    If Intersect(Target, watchedRange) Is Nothing Then Exit Sub

    Application.Run "'" & ThisWorkbook.Name & "'!PassagePlanModule.UpdatePassagePlan", True
End Sub
