Option Explicit

' ============================================================
' PASSAGE PLAN - FIXED3
'
' EXACT RULE:
' Number of pages = CEILING(number of real DATA rows / 35)
'
' DATA rows are counted by ROUTE COLUMN B, starting at row 2.
' Therefore:
' 1-35   -> 1 page
' 36-70  -> 2 pages
' 71-105 -> 3 pages
' 106-140 -> 4 pages
'
' Page layout:
' rows 1-11  = header
' rows 12-46 = 35 data rows
' next page starts at row 47
'
' Mapping:
' DATA A -> Plan list AT
' DATA B -> Plan list G
' DATA C -> Plan list L
' DATA D -> Plan list T
' DATA E -> Plan list Q
' DATA F -> Plan list AB
' DATA G -> Plan list X
' ============================================================

Sub Auto_Open()
    UpdatePassagePlan True
End Sub


Public Sub RouteDataChanged(ByVal Target As Range)

    If Target Is Nothing Then Exit Sub
    If Target.CountLarge > 500 Then Exit Sub
    If Intersect(Target, Target.Worksheet.Range("A2:G600")) Is Nothing Then Exit Sub

    UpdatePassagePlan True

End Sub


Public Sub UpdatePassagePlan(Optional ByVal Silent As Boolean = False)

    Const PLAN_SHEET As String = "Plan list"
    Const DATA_SHEET As String = "DATA"

    Const PAGE_ROWS As Long = 46
    Const DATA_ROWS As Long = 35
    Const TEMPLATE_LAST_ROW As Long = 46
    Const FIRST_DATA_ROW As Long = 12

    Dim wsP As Worksheet
    Dim wsD As Worksheet

    Dim dataCount As Long
    Dim pagesNeeded As Long

    Dim p As Long
    Dim destStart As Long
    Dim oldLastRow As Long
    Dim dataRow As Long
    Dim destRow As Long
    Dim recordIndex As Long

    Dim oldCalc As XlCalculation
    Dim oldEvents As Boolean
    Dim oldScreenUpdating As Boolean

    On Error GoTo ErrHandler

    Set wsP = ThisWorkbook.Worksheets(PLAN_SHEET)
    Set wsD = ThisWorkbook.Worksheets(DATA_SHEET)

    oldCalc = Application.Calculation
    oldEvents = Application.EnableEvents
    oldScreenUpdating = Application.ScreenUpdating

    Application.ScreenUpdating = False
    Application.EnableEvents = False
    Application.Calculation = xlCalculationManual

    dataCount = CountDataRowsByColumnB(wsD)

    If dataCount = 0 Then
        pagesNeeded = 1
    Else
        pagesNeeded = (dataCount + DATA_ROWS - 1) \ DATA_ROWS
    End If

    oldLastRow = LastPlanRow(wsP)

    If oldLastRow > TEMPLATE_LAST_ROW Then
        DeleteShapesBelowRow wsP, TEMPLATE_LAST_ROW
        UnmergeBelowRow wsP, TEMPLATE_LAST_ROW
        wsP.Rows((TEMPLATE_LAST_ROW + 1) & ":" & oldLastRow).Delete
    End If

    wsP.ResetAllPageBreaks

    For p = 2 To pagesNeeded
        destStart = (p - 1) * PAGE_ROWS + 1

        wsP.Rows("1:46").Copy
        wsP.Rows(destStart & ":" & _
                 destStart + PAGE_ROWS - 1).PasteSpecial Paste:=xlPasteAll
        Application.CutCopyMode = False

        CopyRowHeights wsP, 1, 46, destStart
        wsP.Range("C" & CStr(destStart + 4)).Value = p
        CopyTemplateShapes wsP, destStart
    Next p

    ClearMappedCells wsP, pagesNeeded

    recordIndex = 0

    For dataRow = 2 To LastDataRowInColumnB(wsD)
        If HasRealValue(wsD.Cells(dataRow, 2)) Then
            destRow = (recordIndex \ DATA_ROWS) * PAGE_ROWS + _
                      FIRST_DATA_ROW + _
                      (recordIndex Mod DATA_ROWS)

            wsP.Range("AT" & destRow).Formula = _
                "=IFERROR(IF(DATA!A" & dataRow & "=\"\"",\"\"",DATA!A" & dataRow & "),\"\"")"
            wsP.Range("G" & destRow).Formula = _
                "=IFERROR(IF(DATA!B" & dataRow & "=\"\"",\"\"",DATA!B" & dataRow & "),\"\"")"
            wsP.Range("L" & destRow).Formula = _
                "=IFERROR(IF(DATA!C" & dataRow & "=\"\"",\"\"",DATA!C" & dataRow & "),\"\"")"
            wsP.Range("Q" & destRow).Formula = _
                "=IFERROR(IF(DATA!E" & dataRow & "=\"\"",\"\"",DATA!E" & dataRow & "),\"\"")"
            wsP.Range("T" & destRow).Formula = _
                "=IFERROR(IF(DATA!D" & dataRow & "=\"\"",\"\"",DATA!D" & dataRow & "),\"\"")"
            wsP.Range("AB" & destRow).Formula = _
                "=IFERROR(IF(DATA!F" & dataRow & "=\"\"",\"\"",DATA!F" & dataRow & "),\"\"")"
            wsP.Range("X" & destRow).Formula = _
                "=IFERROR(IF(DATA!G" & dataRow & "=\"\"",\"\"",DATA!G" & dataRow & "),\"\"")"

            recordIndex = recordIndex + 1
        End If
    Next dataRow

    wsP.ResetAllPageBreaks
    For p = 1 To pagesNeeded - 1
        wsP.HPageBreaks.Add Before:=wsP.Rows(p * PAGE_ROWS + 1)
    Next p

    wsP.PageSetup.PrintArea = "$A$1:$BN$" & (pagesNeeded * PAGE_ROWS)
    wsP.Calculate

    Application.Calculation = oldCalc
    Application.EnableEvents = oldEvents
    Application.ScreenUpdating = oldScreenUpdating

    If Not Silent Then
        MsgBox "Done." & vbCrLf & vbCrLf & _
               "Filled DATA rows: " & dataCount & vbCrLf & _
               "Pages: " & pagesNeeded, _
               vbInformation, "Passage Plan"
    End If

    Exit Sub

ErrHandler:
    Application.Calculation = oldCalc
    Application.EnableEvents = oldEvents
    Application.ScreenUpdating = oldScreenUpdating
    Application.CutCopyMode = False

    If Not Silent Then
        MsgBox "Error: " & Err.Description, _
               vbExclamation, "Passage Plan"
    End If

End Sub


Private Function CountDataRowsByColumnB(ByVal ws As Worksheet) As Long

    Dim lastRow As Long
    Dim r As Long
    Dim n As Long

    lastRow = LastDataRowInColumnB(ws)

    For r = 2 To lastRow
        If HasRealValue(ws.Cells(r, 2)) Then
            n = n + 1
        End If
    Next r

    CountDataRowsByColumnB = n

End Function


Private Function LastDataRowInColumnB(ByVal ws As Worksheet) As Long

    Dim r As Long

    r = ws.Cells(ws.Rows.Count, "B").End(xlUp).Row

    If r < 2 Then
        LastDataRowInColumnB = 1
    Else
        LastDataRowInColumnB = r
    End If

End Function


Private Function HasRealValue(ByVal cell As Range) As Boolean

    Dim v As Variant
    Dim s As String

    v = cell.Value2

    If IsError(v) Then
        HasRealValue = False
        Exit Function
    End If

    s = Replace$(CStr(v), Chr$(160), " ")
    HasRealValue = Len(Trim$(s)) > 0

End Function


Private Function LastPlanRow(ByVal ws As Worksheet) As Long

    Dim f As Range

    Set f = ws.Cells.Find( _
        What:="*", _
        After:=ws.Range("A1"), _
        LookIn:=xlFormulas, _
        LookAt:=xlPart, _
        SearchOrder:=xlByRows, _
        SearchDirection:=xlPrevious, _
        MatchCase:=False)

    If f Is Nothing Then
        LastPlanRow = 46
    Else
        LastPlanRow = f.Row
    End If

End Function


Private Sub UnmergeBelowRow(ByVal ws As Worksheet, _
                            ByVal firstPageLastRow As Long)

    Dim i As Long
    Dim area As Range
    Dim topRow As Long
    Dim bottomRow As Long

    For i = ws.UsedRange.MergeAreas.Count To 1 Step -1

        Set area = Nothing

        On Error Resume Next
        Set area = ws.UsedRange.MergeAreas(i)
        On Error GoTo 0

        If Not area Is Nothing Then
            topRow = area.Row
            bottomRow = area.Row + area.Rows.Count - 1

            If topRow > firstPageLastRow Or _
               bottomRow > firstPageLastRow Then
                area.UnMerge
            End If
        End If

    Next i

End Sub


Private Sub DeleteShapesBelowRow(ByVal ws As Worksheet, _
                                 ByVal firstPageLastRow As Long)

    Dim i As Long
    Dim r As Long

    For i = ws.Shapes.Count To 1 Step -1
        r = 0

        On Error Resume Next
        r = ws.Shapes(i).TopLeftCell.Row
        On Error GoTo 0

        If r > firstPageLastRow Then
            ws.Shapes(i).Delete
        End If
    Next i

End Sub


Private Sub CopyTemplateShapes(ByVal ws As Worksheet, _
                               ByVal destinationFirstRow As Long)

    Dim originalCount As Long
    Dim i As Long

    Dim shp As Shape
    Dim newShp As Shape
    Dim pastedShapes As ShapeRange
    Dim rowOffset As Double

    originalCount = ws.Shapes.Count
    rowOffset = ws.Rows(destinationFirstRow).Top - ws.Rows(1).Top

    For i = 1 To originalCount
        Set shp = ws.Shapes(i)

        If shp.TopLeftCell.Row >= 1 And _
           shp.TopLeftCell.Row <= 46 Then

            On Error Resume Next

            shp.Copy
            Set pastedShapes = ws.Shapes.Paste
            If Not pastedShapes Is Nothing Then
                Set newShp = pastedShapes.Item(1)
            End If

            If Not newShp Is Nothing Then
                newShp.Top = shp.Top + rowOffset
                newShp.Left = shp.Left
                newShp.Width = shp.Width
                newShp.Height = shp.Height
                newShp.Rotation = shp.Rotation
                newShp.LockAspectRatio = shp.LockAspectRatio
            End If

            Set newShp = Nothing
            Set pastedShapes = Nothing

            On Error GoTo 0
        End If
    Next i

End Sub


Private Sub CopyRowHeights(ByVal ws As Worksheet, _
                           ByVal sourceFirstRow As Long, _
                           ByVal sourceLastRow As Long, _
                           ByVal destinationFirstRow As Long)

    Dim r As Long
    Dim d As Long

    For r = sourceFirstRow To sourceLastRow
        d = destinationFirstRow + r - sourceFirstRow
        ws.Rows(d).RowHeight = ws.Rows(r).RowHeight
    Next r

End Sub


Private Sub ClearMappedCells(ByVal ws As Worksheet, _
                             ByVal pagesNeeded As Long)

    Const PAGE_ROWS As Long = 46
    Const FIRST_DATA_ROW As Long = 12
    Const DATA_ROWS As Long = 35

    Dim p As Long
    Dim r As Long
    Dim targetRow As Long

    For p = 0 To pagesNeeded - 1
        For r = 0 To DATA_ROWS - 1
            targetRow = p * PAGE_ROWS + FIRST_DATA_ROW + r

            ws.Range("G" & targetRow).ClearContents
            ws.Range("L" & targetRow).ClearContents
            ws.Range("Q" & targetRow).ClearContents
            ws.Range("T" & targetRow).ClearContents
            ws.Range("X" & targetRow).ClearContents
            ws.Range("AB" & targetRow).ClearContents
            ws.Range("AT" & targetRow).ClearContents
        Next r
    Next p

End Sub
